declare -a s=(); declare -A p=([')']='(' [']']='[' ['}']='{')
for ((i=0; i<${#1}; i++)); do c="${1:i:1}"
  [[ $c == ['([{'] ]] && s+=("$c")
  [[ $c == [')]}'] ]] && { ((!${#s[@]})) || [[ ${s[-1]} != ${p[$c]} ]] && { echo false; exit 0; }; unset 's[-1]'; }
done
((!${#s[@]})) && echo true || echo false
